include(${PROJECT_SOURCE_DIR}/cmake/RutMaxMindDB.cmake)

add_executable(rut-geoip main.cc)
target_link_libraries(rut-geoip PRIVATE maxminddb::maxminddb rut_integer_parser)
if(NOT RUT_SANITIZE)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_link_options(rut-geoip PRIVATE -nostdlib++)
    elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        set_target_properties(rut-geoip PROPERTIES LINKER_LANGUAGE C)
    endif()
endif()
if(RUT_STATIC_RUNTIME)
    target_link_options(rut-geoip PRIVATE -static)
endif()
